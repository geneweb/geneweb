%( H: one row per union, the cells of the person spanning them;
   I: one row for the descendant, then one row per spouse %)
%define;one_child(prefx, fff, ch_cnt)
  %let;up;prefx%in;
  %let;labl;%if;("prefx"!="")prefxfff.%end;ch_cnt%in;
  %let;seen;%apply;mem_sorted_list("listc", index)%in;
  %let;rowspan;%if;(e.t="H" and nb_families>1 and famcols=1) rowspan="%nb_families;"%end;%in;
  %if;(e.t="H" and has_families)
    %foreach;family;
      %if;((seen=0 or e.implx="") and (family_cnt=1 or famcols=1))
        <tr>
          %if;(family_cnt=1)
            %apply;num_cell(seen+mem_sorted_list("listb", family.index))
            %apply;name_cell()
            %apply;head_cells("self")
          %end;
          %apply;fam_cells("")
          %if;(family_cnt=1)%apply;tail_cells("self")%end;
        </tr>
      %end;
      %apply;add_in_sorted_listb(family.index)
    %end;
  %else;
    %if;(seen=0 or e.implx="")
      <tr>
        %apply;num_cell(seen)
        %apply;name_cell()
        %apply;head_cells("self")
        %if;(nb_families=0)
          %if;(e.marr="on" and e.t="H")<td%if;(e.hl=1) class="bg-lg"%end;></td>%end;
          %if;(e.marr_date="on")<td%if;(e.hl=1) class="bg-lg"%end;></td>%end;
          %if;(e.marr_place="on")<td%if;(e.hl=1) class="bg-lg"%end;></td>%end;
          %if;(e.child="on")<td%if;(e.hl=1) class="bg-lg"%end;></td>%end;
        %elseif;(nb_families=1)
          %foreach;family;%apply;fam_cells(2)%end;
        %else;
          %if;(e.marr_date="on")<td%if;(e.hl=1) class="bg-lg"%end; title="[*edit family on spouse]"></td>%end;
          %if;(e.marr_place="on")<td%if;(e.hl=1) class="bg-lg"%end;></td>%end;
          %if;(e.child="on")<td class="align-middle text-center">%nb_children_total;</td>%end;
        %end;
        %apply;tail_cells("self")
      </tr>
    %end;
    %if;(e.t="I" and famcols=1)
      %foreach;family;
        %let;imp;%expr(mem_sorted_list("listb", family.index)+mem_sorted_list("listc", spouse.index))%in;
        %if;(imp<2 or e.implx="")%apply;spouse_row(imp)%end;
        %apply;add_in_sorted_listb(family.index)
      %end;
    %end;
  %end;
%end;

%define;families_of_one_p(prefx)
  %( Les enfants d’une personne sont numérotés de 1 à n indépendamment du nombre de mariages %)
  %foreach;family;
    %let;fam;%if;(nb_families>1)%apply;letter(family_cnt)%end;%in;
    %let;off;%nb_children_before;%in;
    %foreach;child;
      %apply;one_child("prefx", fam, off+child_cnt)
    %end;
  %end;
%end;

%( Parcours de l'arbre descendant, niveau par niveau %)
%define;one_person(xxx, lv, max_l, prefx, fam_cnt, ch_cnt)
  %if;(lv<=max_l)
    <div>
      %apply;list_row("xxx", "prefx", fam_cnt, ch_cnt)
    </div>
    %if;xxx.has_families;
      %foreach;family;
        %let;fam;%if;(nb_families>1)%apply;letter(family_cnt)%end;%in;
        %let;off;%nb_children_before;%in;
        %foreach;child;
          %let;npref;prefx%fam;.%expr(off+child_cnt)%in;
          %apply;one_person("child", lv+1, max_l, npref, family_cnt, child_cnt)
        %end;
      %end;
    %end;
  %end;
%end;

%( Parcours de l'arbre, horizontal, niveau par niveau descendant %)
%( À chaque niveau, prefx et faml (lettre) définissent le préfixe d'Aboville %)
%define;one_level(lv, max_l, prefx, faml, ch_cnt)
  %let;npref;%if;("prefx"!="")prefxfaml.%end;ch_cnt%in;
  %if;(lv<max_l)
    %foreach;family;
      %let;fam;%if;(nb_families>1)%apply;letter(family_cnt)%end;%in;
      %let;off;%nb_children_before;%in;
      %foreach;child;
        %apply;one_level(lv+1, max_l, npref, fam, off+child_cnt)
      %end;
    %end;
  %else;
    %apply;families_of_one_p(npref)
  %end;
%end;
%define;gen_count(p_lev)

        %if;(nb=1)[person/persons]0%else;[person/persons]1%end;
      %elseif;(nb=1)[cousins.0.p_lev]0
      %else;[cousins.0.p_lev]1
      %end;
    %end;
  %end;
%end;
