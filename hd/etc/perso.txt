<script>
function reducePlaces() {
  const seen = [];
  document.querySelectorAll('[data-updfield$="_place"]').forEach(el => {
    const a = el.querySelector('abbr');
    const full = (a ? a.title : el.textContent).trim();
    const raw = (el.dataset.updraw || '').trim();
    const sub = raw.startsWith('%[') && /\%]\s(?:–|—|-)\s/.test(raw);
    const sep = sub ? full.search(/\s(?:–|—|-)\s/) : -1;
    const i = full.indexOf(',', sep < 0 ? 0 : sep);
    if (i > 0 &&
        seen.some(s => full.endsWith(s) && full.length - s.length <= i)) {
      const abbr = document.createElement('abbr');
      abbr.title = full;
      abbr.textContent = full.slice(0, i);
      el.replaceChildren(abbr);
    } else if (a) {
      el.textContent = full;
    }
    seen.push(full);
  });
}
reducePlaces();
</script>

<script>
%( Hide what the time line already shows: an element whose data-dup key is
   also found in #timeline is hidden (its whole list item inside a .dup-box),
   then any .dup-box left without a visible item. %)
function hideTimelineDuplicates() {
  const timeline = document.getElementById('timeline');
  if (!timeline) return;
  const keys = new Set();
  timeline.querySelectorAll('%[data-dup%]').forEach(el => keys.add(el.dataset.dup));
  document.querySelectorAll('%[data-dup%]').forEach(el => {
    if (timeline.contains(el) || !keys.has(el.dataset.dup)) return;
    const target = el.closest('.dup-box li') || el;
    target.hidden = true;
  });
  document.querySelectorAll('%[data-dup-alt%]').forEach(el => {
    if (keys.has(el.dataset.dupAlt)) el.hidden = false;
  });
  document.querySelectorAll('.dup-box').forEach(box => {
    const items = Array.from(box.querySelectorAll(':scope > ul > li'));
    if (items.every(li => li.hidden)) box.hidden = true;
  });
}
hideTimelineDuplicates();
%( The time line toggles show or hide their rows in place and keep the
   state in the URL, so that a reload gives the same rows. %)
function setupTimelineToggles() {
  document.querySelectorAll('%[data-tl-toggle%]').forEach(btn => {
    btn.addEventListener('click', () => {
      const key = btn.dataset.tlToggle;
      const on = btn.getAttribute('aria-pressed') !== 'true';
      btn.setAttribute('aria-pressed', on);
      btn.classList.toggle('text-body-secondary', !on);
      btn.title = on ? btn.dataset.hide : btn.dataset.show;
      document.querySelectorAll(`#timeline tr%[data-tl="${key}"%]`).forEach(tr => {
        tr.hidden = !on;
      });
      const url = new URL(location.href);
      if (on) url.searchParams.set(key, 'on');
      else url.searchParams.delete(key);
      history.replaceState(null, '', url);
    });
  });
}
setupTimelineToggles();
</script>
