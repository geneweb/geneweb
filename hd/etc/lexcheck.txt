<script>
(function () {
  function run() {
    var OPEN = String.fromCharCode(91), miss = 0;
    var t = document.getElementById('lexcheck');
    if (!t) return;
    var lines = t.querySelectorAll('tbody td div');
    for (var i = 0; i < lines.length; i++) {
      if (lines[i].textContent.indexOf(OPEN) > -1) {
        lines[i].classList.add('bg-warning-subtle');
        miss++;
      }
    }
    var c = document.getElementById('lexcheck-miss');
    if (c) c.textContent = miss;
    
    /* epicene retombe sur le masculin */
    var tds = t.querySelectorAll('tbody td');
    for (var k = 0; k < tds.length; k++) {
      var d = tds[k].querySelectorAll('div');
      if (d.length >= 6 && d[3].textContent.trim() === d[5].textContent.trim())
        d[5].classList.add('bg-danger-subtle');
    }
  }
  
  if (document.readyState === 'loading')
    document.addEventListener('DOMContentLoaded', run);
  else run();
}());
</script>
