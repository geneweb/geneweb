%( Date (dd/mm/yyyy in [!dates order]) and place of a created person.
   kind: birth or death; xvar: field prefix; xx: template variable. %)
%define;small_date(kind,xvar,xx)
  %let;day_input;pattern="(?:0?[1-9]|1[0-9]|2[0-9]|3[0-1])" size="2" maxlength="2"%in;
  %let;month_input;pattern="(?:0?[1-9]|1[0-2]|VD|BR|FM|NI|PL|VT|GE|FL|PR|ME|TH|FT|JC|vd|br|fm|ni|pl|vt|ge|fl|pr|me|th|ft|jc)" size="1" maxlength="2"%in;
  %let;year_input;pattern="[?><~\/+\-]?\d*\/?" size="4" maxlength="8"%in;
  <div class="row align-items-center">
    <span class="col-sm-2 col-form-label">[*kind]</span>
    <div class="col-sm-auto d-inline-flex">
      %if;([!dates order]0="ddmmyy" or [!dates order]0="ddmmyyyy" or [!dates order]0="dmyyyy")
        <label for="xvar_dd" class="visually-hidden">[year/month/day]2</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_dd" id="xvar_dd" value="%xx.create.kind_day;" placeholder="[dd/mm/yyyy]0" %day_input;>
        <label for="xvar_mm" class="visually-hidden">[year/month/day]1</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_mm" id="xvar_mm" value="%xx.create.kind_month;" placeholder="[dd/mm/yyyy]1" %month_input;>
        <label for="xvar_yyyy" class="visually-hidden">[year/month/day]0</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_yyyy" id="xvar_yyyy" value="%xx.create.kind_year;" placeholder="[dd/mm/yyyy]2" %year_input;>
      %elseif;([!dates order]0="mmddyyyy")
        <label for="xvar_mm" class="visually-hidden">[year/month/day]1</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_mm" id="xvar_mm" value="%xx.create.kind_month;" placeholder="[dd/mm/yyyy]1" %month_input;>
        <label for="xvar_dd" class="visually-hidden">[year/month/day]2</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_dd" id="xvar_dd" value="%xx.create.kind_day;" placeholder="[dd/mm/yyyy]0" %day_input;>
        <label for="xvar_yyyy" class="visually-hidden">[year/month/day]0</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_yyyy" id="xvar_yyyy" value="%xx.create.kind_year;" placeholder="[dd/mm/yyyy]2" %year_input;>
      %else;
        <label for="xvar_yyyy" class="visually-hidden">[year/month/day]0</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_yyyy" id="xvar_yyyy" value="%xx.create.kind_year;" placeholder="[dd/mm/yyyy]2" %year_input;>
        <label for="xvar_mm" class="visually-hidden">[year/month/day]1</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_mm" id="xvar_mm" value="%xx.create.kind_month;" placeholder="[dd/mm/yyyy]1" %month_input;>
        <label for="xvar_dd" class="visually-hidden">[year/month/day]2</label>
        <input type="text" class="form-control form-control-inline me-2" name="xvar_dd" id="xvar_dd" value="%xx.create.kind_day;" placeholder="[dd/mm/yyyy]0" %day_input;>
      %end;
    </div>
    <label for="xvar_pl" class="visually-hidden">[place/places]0</label>%nn;
    <div class="col">
      <input type="text" class="form-control" name="xvar_pl" value="%xx.create.kind_place;" id="xvar_pl" placeholder="[*place/places]0"%datalist_places;>
    </div>
  </div>
%end;

%( Fields of a person created from an update form (parent, child, witness,
   related person), shown when its selector is on "create": the wrapper id
   [xvar]_p_selct_data is toggled by initSelectToggle.
   xvar: prefix of the fields (pa1, ch2, e1_witn3, r1_fath), read back by
   Update_util.reconstitute_create_info; xx: template variable of the person. %)
%define;create_data(xvar,xx)
  <div id="xvar_p_selct_data" class="mt-2">
    %apply;small_date("birth","xvarb","xx")
    %apply;small_date("death","xvard","xx")
    <div class="row">
      <label for="xvar_occu" class="col-sm-2 col-form-label">[*occupation/occupations]0</label>
      <div class="col-sm-6">
        <input class="form-control" type="text" name="xvar_occu" id="xvar_occu"
          value="%xx.create.occupation;" placeholder="[*occupation/occupations]0"%datalist_occupations;>
      </div>
      <div class="col-sm-auto d-inline-flex align-items-center">
        <div class="form-check form-check-inline me-3">
          <input class="form-check-input" type="checkbox" name="xvar_od" id="xvar_od"%if;xx.create.of_course_dead; checked%end;>
          <label for="xvar_od" class="form-check-label">[*of course dead]</label>
        </div>
        <div class="form-check form-check-inline">
          <input class="form-check-input" type="checkbox" name="xvar_pub" id="xvar_pub"%if;xx.create.public; checked%end;>
          <label for="xvar_pub" class="form-check-label">[*public]</label>
        </div>
      </div>
    </div>
  </div>
%end;
