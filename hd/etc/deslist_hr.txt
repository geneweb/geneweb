%( Union line after its prefix: the person, the union and the spouse %)
%define;union_line()
  %if;(family_cnt>1 or cancel_links)
    %self; %dates_notag;%if;(title!=""), %title;%end;%nn;
  %else;
    <a href="%prefix;%access;" data-index="%index;">%self;</a>
    %if;wizard;<a href="%prefix;m=MOD_IND&i=%index;" title="[modify::person/persons]0">%dates_notag;</a>%else;%dates_notag;%end;
    %if;(title!=""), %title;%end;
  %end;%sp;
  %if;(wizard and not cancel_links)
    <a href="%prefix;m=MOD_FAM&i=%family.index;&ip=%index;" title="[modify::family/families]0">%nn;
  %end;
    <span class="small text-body-secondary text-nowrap fst-italic">%nn;
      %if;(are_engaged or are_married)&%elseif;are_not_married;X%end;
      %if;(on_marriage_date!="")%family.date_s;%end;
    </span>%nn;
  %if;(wizard and not cancel_links)</a>%end;%sp;
  %if;cancel_links;%spouse; %spouse.dates_notag;%if;(spouse.title!=""), %spouse.title;%end;%nn;
  %else;
    <a href="%prefix;%spouse.access;">%spouse;</a>
    %if;wizard;<a href="%prefix;m=MOD_IND&i=%spouse.index;" title="[modify::person/persons]0">%spouse.dates_notag;</a>%else;%spouse.dates_notag;%end;
    %if;(spouse.title!=""), %spouse.title;%end;%nn;
  %end;
  %if;spouse.is_descendant;
    %if;(e.implx="")%apply;implex_btn(spouse.index)%end;
  %elseif;(spouse.first_name!="?" and spouse.surname!="?")
    %apply;add_in_sorted_list(spouse.index)
  %end;
%end;

%( A person of the tree, one line per union, then the children of each union.
   p_pre: line start up to the child number, p_ind: column under the person,
   p_lev: generation %)
%define;desc1(p_pre,p_ind,p_lev)
  %let;seen;%apply;mem_sorted_list("listc", index)%in;
  %let;num;%if;(p_lev!=0)%if;cancel_links;%child_cnt;%else;<a href="%prefix;%access;&m=D&t=D&v=%expr(l_v-p_lev)" title="[navigate on tree] [with] %self;">%child_cnt;</a>%end;%end;%in;
  %apply;add_in_sorted_list(index)
  %apply;add_in_sorted_listc(index)
  %if;has_families;
    %let;hv;%if;is_male;1%end;%in;
    %let;smark;%if;(browsing_with_sosa_ref and has_sosa)%if;has_siblings;▶%else;▷%end;%end;%in;
    %foreach;family;
      <span data-surname="_%surname;" class="text-nowrap"><span class="tt">%nn;
        %if;(family_cnt=1 or p_lev=0)p_pre%num;%else;p_ind%if;(child_cnt<10) %end;%child_cnt;%end;%nn;
        %if;(nb_families>1)%apply;letter(family_cnt)%elseif;(hv=1)━%else;─%end;%nn;
        %if;has_children;%if;(hv=1)┳%else;┬%end;%elseif;(hv=1)━%else;─%end;%smark;</span>
        %apply;union_line()
      </span>
      <br data-surname="_%surname;">
      %if;(p_lev<l_v)
        %let;l1;%if;(family_cnt=nb_families)   %else;  ┊%end;%in;
        %let;fseen;%if;(e.implx!="")%apply;mem_sorted_list("listb", family.index)%end;%in;
        %foreach;child;
          %let;lst;%if;(child_cnt=nb_children)1%end;%in;
          %let;pre;p_ind%l1;%if;(hv=1)%if;(lst=1)┗%else;┣%end;%if;(child_cnt<10)━%end;%else;%if;(lst=1)└%else;├%end;%if;(child_cnt<10)─%end;%end;%in;
          %if;(fseen=1 and mem_sorted_list("listc", index)=1)%apply;desc2(pre)
          %else;
            %let;ind;p_ind%l1;%if;(lst=1) %elseif;(hv=1)┃%else;│%end;%in;
            %apply;desc1(pre, ind, p_lev+1)
          %end;
        %end;
      %end;
      %apply;add_in_sorted_listb(family.index)
    %end;
  %else;
    %let;tail;%if;is_male;━━%elseif;is_female;──%else;╍%end;%in;
    <span data-surname="_%surname;" class="text-nowrap"><span class="tt">p_pre%num;%tail;%nn;
      %if;(browsing_with_sosa_ref and has_sosa)%if;has_siblings;◀%else;◁%end;%end;%if;(seen=1)%tail;%end;</span>
      %if;cancel_links;%self;%else;<a href="%prefix;%access;" data-index="%index;">%self;</a>%if;(title!=""), %title;%end;%end;%dates;
    </span>
    <br data-surname="_%surname;">
  %end;
%end;
