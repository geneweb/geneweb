%( m=A ancestor table by couple t=Y + &t=X and table upto an ancestor %)
%( 1 when the couple row holding sosa p_s is displayed %)
%define;row_shown(p_s)%if;(e.t="X")%apply;mem_sorted_list("listb", p_s/2)%elseif;(p_s>=2^lmin and p_s<2^(l_v+1))1%else;0%end;%end;

%define;num_cell()
  %let;fs;%if;(only=0)%ancestor.father_sosa;%end;%in;
  <td id="i%ancestor.anc_sosa.v;" class="align-middle text-center">
    %if;cancel_links;<strong>%ancestor.anc_sosa;</strong>
    %elseif;(fs!="" and row_shown(fs)=1)
      <a href="#i%fs;" title="[father/mother]0[:] %apply;1000sep(fs)"><strong>%ancestor.anc_sosa;</strong></a>
    %elseif;(wizard and not ancestor.has_parents and ancestor.first_name!="?" and ancestor.surname!="?")
      <a class="text-danger" href="%prefix;m=ADD_PAR&ip=%ancestor.index;" target="_blank"
        title="[*add::parents]"><strong>%ancestor.anc_sosa;</strong></a>
    %else;<strong>%ancestor.anc_sosa;</strong>
    %end;
    <br>
    %if;(ancestor.same="")
      %if;not cancel_links;
        <a href="%apply;url_set%with;t/i1/l%and;X/%ancestor.index;/%level;%end;"
          title="[*only] [ancestor/ancestors]1 [up to] %ancestor;">%nn;
      %end;
      %if;ancestor.has_siblings;&diams;%elseif;ancestor.has_parents;&loz;%else;-%end;%nn;
      %if;not cancel_links;</a>%end;
      %if;(ancestor.nb_families>1)
        <div class="small">
          <span title="%ancestor.nb_families; [relation/relations]1">%ancestor.nb_families; &amp;</span>
          <span title="[*number of children]">%ancestor.nb_children;%nn;
            <i class="fa fa-child fa-sm ms-1" aria-hidden="true"></i></span>
        </div>
      %end;
    %else;
      %if;cancel_links;==
      %else;<a href="%prefix;em=R&ei=%ancestor.index;&m=NG&select=%index;&n=1&t=PN&et=A"
        title="%ancestor; …[relationship]… %self;">==</a>
      %end;
      <br>
      %if;(not cancel_links and row_shown(ancestor.same.v)=1)
        <a href="#i%ancestor.same.v;" title="[*implex/implexes]0">%ancestor.same;</a>
      %else;%ancestor.same;
      %end;
    %end;
  </td>
%end;

%define;pers_cell(xx,z1)
  <td class="align-middle text-center%if;("z1"!="") z1%end;">
    %if;(e.im!=0 and e.image!="off" and xx.has_image)
      <img class="img-fluid w-50" src="%xx.image_url;" alt="%xx;">%nn;
    %end;
    <div>%if;cancel_links;%xx;%else;<a href="%prefix;%xx.access;">%xx;</a>%end;</div>
    %if;xx.has_nobility_titles;<div class="fst-italic">%xx.nobility_title;</div>%end;
    <div class="small">
      %if;xx.has_occupation;<div>%xx.occupation;</div>%end;
      %if;xx.computable_age;<div>[age][:] %xx.age;</div>%end;
      %if;xx.computable_death_age;<div title="[*age at death]">%xx.death_age;</div>%end;
      %if;(xx.has_consanguinity and (b.show_consang!="no" or manitou))
        <div title="[*consanguinity]">%xx.consanguinity;</div>
      %end;
    </div>
  </td>
%end;

%( Birth (or baptism) and death (or burial) date and place %)
%define;dates_cell(xx)
  %let;bd;%if;(xx.birth_date.year!="")%xx.slash_birth_date;%elseif;(xx.baptism_date.year!="")%xx.slash_baptism_date;%end;%in;
  %let;bp;%if;xx.has_birth_place;%xx.birth_place;%elseif;xx.has_baptism_place;%xx.baptism_place;%end;%in;
  %let;bi;%if;(e.hl=1 and (bd="" or bp=""
    or (xx.has_birth_date and (xx.birth_date.prec!="" or xx.birth_date.day="" or xx.birth_date.month=""))
    or (not xx.has_birth_date and (xx.baptism_date.prec!="" or xx.baptism_date.day="" or xx.baptism_date.month=""))))1%end;%in;
  %let;dd;%if;(xx.death_date.year!="")%xx.slash_death_date;%elseif;(xx.burial_date.year!="")%xx.slash_burial_date;%end;%in;
  %let;dp;%if;xx.has_death_place;%xx.death_place;%elseif;xx.has_burial_place;%xx.burial_place;%end;%in;
  %let;di;%if;(e.hl=1 and (dd="" or dp=""
    or (xx.has_death_date and (xx.death_date.prec!="" or xx.death_date.day="" or xx.death_date.month=""))
    or (not xx.has_death_date and (xx.burial_date.prec!="" or xx.burial_date.day="" or xx.burial_date.month=""))))1%end;%in;
  <td class="align-middle text-start table_big_cell">
    %if;(wizard and not cancel_links)
      <a class="date table_big_anchor" href="%prefix;m=MOD_IND&i=%xx.index;" target="_blank"
        title="[*modify::person/persons]0">%nn;
    %end;
    %if;(bd!="" or bp!="")
      <span%if;(bi=1) class="bg-ly-incomplete"%end;>%nn;
        <span class="font-monospace">%birth_symbol;</span>%bd;%if;(bp!="") %bp;%end;</span>%nn;
    %end;
    %if;(xx.is_dead and (dd!="" or dp!=""))
      %if;(bd!="" or bp!="")<br>%end;
      <span%if;(di=1) class="bg-ly-incomplete"%end;>%nn;
        <span class="font-monospace">%death_symbol;</span>%dd;%if;(dp!="") %dp;%end;</span>%nn;
    %end;
    %if;(wizard and not cancel_links)</a>%end;
  </td>
%end;

%define;month(mmm)
  %if;(mmm.month="")%if;(mmm.prec="&gt;")12%else;00%end;
  %elseif;(mmm.month<10)0%mmm.month;%else;%mmm.month;%end;
%end;

%define;day(ddd)
  %if;(ddd.day="")%if;(ddd.prec="&gt;")32%else;00%end;
  %elseif;(ddd.day<10)0%ddd.day;%else;%ddd.day;%end;
%end;

%( One marriage of the couple cell: yy=1 for the couple itself, 0 for another union %)
%define;link_marriage(yy)
  %if;(wizard and not cancel_links)
    <a class="date%if;(yy=1) table_big_anchor%end;" href="%prefix;m=MOD_FAM&i=%family.index;&ip=%ancestor.index;" target="_blank">%nn;
  %end;
  <div class="%if;(yy=0)small %if;ancestor.is_male;male%elseif;ancestor.is_female;female%end;%end;%nn;
    %if;(e.hl=1 and (slash_marriage_date="" or marriage_date.prec!="" or marriage_date.day="" or marriage_date.month="" or marriage_place="")) not_sure_dates bg-ly-incomplete%end;"
    title="%if;(wizard and not cancel_links)[*modify::family/families]0%end;
      %if;(yy=0 and spouse.surname!="?" and spouse.first_name!="?")&#010;
        %if;(are_not_married)([*not married])&#010;%end;
        %if;(ancestor.is_male)[*him/her]0 %if;(are_not_married)[with]%else;[married%t to:::]0%end;
        %elseif;(ancestor.is_female)[*him/her]1 %if;(are_not_married)[with]%else;[married%t to:::]1%end;
        %end; %spouse;&#010;
        %if;(nb_children>0)%nb_children; %end;%if;(nb_children=0)[*no descendants]%elseif;(nb_children>1)[child/children]1%else;[child/children]0%end;%end;">%nn;
    %nb_children;· %slash_marriage_date; %marriage_place;%if;are_not_married; –%end;</div>%nn;
  %if;(wizard and not cancel_links)</a>%end;
%end;

%define;addlist(yy)
  %apply;add_in_sorted_list%with;
    %if;(e.rev=1)-%end;%marriage_date.year;%nn;
    %and;%apply;month("marriage_date")%nn;
    %and;%apply;day("marriage_date")%nn;
    %and;%apply;link_marriage(yy)%end;
%end;

%( A couple row is opened by the father (even sosa) and closed by the mother;
   the marriages of both are listed in the middle cell, sorted by date %)
%define;one_line(z1)
  %let;odd;%expr(ancestor.anc_sosa.v%2)%in;
  %if;(odd=0)
    %reset_count1;%if;(ancestor.same!="")%incr_count1;%end;
    <tr>
      %apply;num_cell()
      %apply;pers_cell("ancestor","z1")
      %apply;dates_cell("ancestor")
      <td class="align-middle text-start table_big_cell">
  %end;
  %if;(ancestor.same="")
    %foreach;ancestor.family;
      %if;(family.index!=ancestor.family.index)%apply;addlist(0)
      %elseif;(odd=0 or count1=1)%apply;addlist(1)
      %end;
    %end;
  %end;
  %if;(odd=1)
    %foreach;sorted_list_item;%item.4;%end;
    %empty_sorted_list;
      </td>
      %apply;dates_cell("ancestor")
      %apply;pers_cell("ancestor","z1")
      %apply;num_cell()
    </tr>
  %end;
%end;

%( t=X: couples between the sosa 1 and the ancestor i1 at level l %)
%define;table_to_ancestor()
  %pvar.1.mark_descendants;
  %empty_sorted_listb;
  %foreach;ancestor_level(e.l)
    %foreach;ancestor;
      %if;(level>0 and ancestor.is_descendant)%apply;add_in_sorted_listb(ancestor.anc_sosa.v/2)%end;
    %end;
  %end;
  %foreach;ancestor_level(e.l)
    %if;(level>0)
      %apply;lazy_print%with;
        <tbody>
        %if;(e.gen=1)
          <tr><th scope="rowgroup" colspan="7" class="text-center gen-row">[*generation/generations]0 %level;</th></tr>
        %end;
      %end;
      %foreach;ancestor;
        %if;(mem_sorted_list("listb", ancestor.anc_sosa.v/2)=1)
          %lazy_force;
          %apply;one_line%with;%if;ancestor.is_descendant;side-border-%if;ancestor.is_female;fe%end;male%end;%end;
        %end;
      %end;
      %if;lazy_printed;</tbody>%end;
    %end;
  %end;
%end;

%( t=Y: couples of the generations lmin to l_v %)
%define;table_double()
  %foreach;ancestor_level(l_v)
    %if;(level>=lmin)
      <tbody>
        %if;(e.gen=1 and level>1)
          <tr><th scope="rowgroup" colspan="7" class="text-center gen-row">[*generation/generations]0 %level;</th></tr>
        %end;
        %foreach;ancestor;
          %if;(ancestor.same="" and ancestor.first_name!="?" and ancestor.surname!="?")%incr_count;%end;
          %apply;one_line("")
        %end;
      </tbody>
    %end;
  %end;
  <tfoot>
    <tr><td colspan="7" class="text-center">%nn;
      [*total][:] %count;/%expr(2^(l_v+1)-2^lmin) [ancestor/ancestors]n.</td>
    </tr>
  </tfoot>
%end;
