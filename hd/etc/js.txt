function insertTags(tagOpen, tagClose, sampleText) {
  const ta = document.querySelector('textarea#notes_comments');
  if (!ta) return false;
  const { selectionStart: start, selectionEnd: end, scrollTop } = ta;
  let sel = ta.value.substring(start, end) || sampleText;
  %( Keep a trailing space of the selection outside the closing tag. %)
  const trail = sel.endsWith(' ') ? ' ' : '';
  if (trail) sel = sel.slice(0, -1);
  ta.setRangeText(tagOpen + sel + tagClose + trail, start, end);
  ta.scrollTop = scrollTop;
  const pos = start + tagOpen.length;
  if (start === end) {
    ta.setSelectionRange(pos, pos + sampleText.length);
  } else {
    const after = pos + sel.length + tagClose.length;
    ta.setSelectionRange(after, after);
  }
  ta.focus();
  return false;
}

%( Shared submit button for the person and title search forms. %)
function initializeWelcomeSearchFunctionality() {
  const mainForm = document.getElementById('main-search');
  const titleForm = document.getElementById('title-search');
  const searchBtn = document.getElementById('global-search');
  if (!searchBtn) return;

  const defaultText = `[*search/case sensitive]0`;

  function hasInput(form) {
    return form && Array.from(form.querySelectorAll('input[type="search"]'))
                      .some(input => input.value.trim() !== '');
  }

  function closeAlertsAndCleanURL() {
    document.querySelectorAll('.alert').forEach(el => {
      bootstrap.Alert.getOrCreateInstance(el).close();
    });
    const url = new URL(window.location.href);
    const bParam = url.searchParams.get('b');
    url.search = '';
    if (bParam !== null) url.searchParams.set('b', bParam);
    window.history.replaceState({}, '', url);
  }

  function updateTooltip() {
    const instance = bootstrap.Tooltip.getInstance(searchBtn);
    if (!instance) return;
    let text = defaultText;
    if (hasInput(mainForm)) text = `[*search/case sensitive]0 [person/persons]0`;
    else if (titleForm && hasInput(titleForm)) text = `[*search/case sensitive]0 [title/titles]0/[estate]0`;
    instance.setContent({ '.tooltip-inner': text });
    if (text !== defaultText) {
      instance.show();
    } else {
      instance.hide();
    }
  }

  [mainForm, titleForm].forEach(form => {
    if (!form) return;
    form.querySelectorAll('input[type="search"]').forEach(input => {
      input.addEventListener('input', () => {
        if (hasInput(form)) closeAlertsAndCleanURL();
        updateTooltip();
      });
    });
    form.addEventListener('keypress', e => {
      if (e.key === 'Enter') {
        e.preventDefault();
        if (hasInput(form)) form.submit();
      }
    });
  });

  searchBtn.addEventListener('click', () => {
    if (hasInput(mainForm)) mainForm.submit();
    else if (titleForm && hasInput(titleForm)) titleForm.submit();
  });
}
document.addEventListener('DOMContentLoaded', initializeWelcomeSearchFunctionality);

    %end;
  %].forEach(t => {
    const dl = document.getElementById('datalist_' + t);
    if (!dl) return;
    fetch(path + t + '.cache.gz')
      .then(r => {
        if (!r.ok) throw new Error(r.status);
        return new Response(r.body.pipeThrough(new DecompressionStream('gzip'))).text();
      })
      .then(text => {
        const frag = document.createDocumentFragment();
        for (const line of text.split('\n')) {
          const v = line.trim().replace('<option>', '');
          if (v) frag.appendChild(new Option('', v));
        }
        dl.replaceChildren(frag);
      })
      .catch(err => console.error(`datalist_${t}:`, err));
  });
}

      return false;
    }

    try {
      this.client = new Rpc(this.RPC_URL);
      this.client.lookup = async (...args) => {
        const jargs = args.map(a => JSON.stringify(a));
        const response = await this.client.call('lookup', jargs);
        return JSON.parse(response);
      };

      %( Probe the server, give up after 3 s. %)
      const testPromise = this.client.lookup('%base.name;_fnames.cache', '', 1);
      const timeoutPromise = new Promise((_, reject) =>
        setTimeout(() => reject(new Error('RPC timeout')), 3000)
      );
      await Promise.race(%[testPromise, timeoutPromise%]);

      this.isConnected = true;
      console.log('RpcDatalistManager: Connecté au serveur RPC');
      return true;
    } catch (error) {
      console.warn('RpcDatalistManager: Serveur RPC indisponible, fallback .gz', error);
      this.isConnected = false;
      return false;
    }
  },

  populateDatalist: async function(datalistId, searchTerm) {
    if (!this.isConnected || !this.client) return;

    const dataList = document.getElementById(datalistId);
    if (!dataList) return;

    if (!searchTerm || searchTerm.length < 3) {
      dataList.innerHTML = '';
      return;
    }

    try {
      const cacheFile = this.getCacheFile(datalistId);
      const results = await this.client.lookup(cacheFile, searchTerm, 50);

      const fragment = document.createDocumentFragment();
      results.forEach(item => {
        const opt = document.createElement('option');
        opt.value = item;
        fragment.appendChild(opt);
      });

      dataList.innerHTML = '';
      dataList.appendChild(fragment);
    } catch (error) {
      console.error(`RpcDatalistManager: Erreur lookup ${datalistId}`, error);
    }
  },

  %( Bind every input whose list attribute starts with datalist_. %)
  attachAllListeners: function() {
    const self = this;
    document.querySelectorAll('input[list^="datalist_"]').forEach(input => {
      const datalistId = input.getAttribute('list');

      input.addEventListener('input', function() {
        clearTimeout(self.debounceTimers[datalistId]);
        self.debounceTimers[datalistId] = setTimeout(() => {
          self.populateDatalist(datalistId, this.value);
        }, 100);
      });

      input.addEventListener('focus', function() {
        if (this.value && this.value.length >= 2) {
          self.populateDatalist(datalistId, this.value);
        }
      });
    });
  }
};
document.addEventListener('DOMContentLoaded', async () => {
  if (await RpcDatalistManager.init()) {
    RpcDatalistManager.attachAllListeners();
  } else if (typeof populateDatalists === 'function') {
    %( .gz fallback when the RPC server is unavailable %)
    populateDatalists();
  }
});
</script>

%( Floating placeholders for textual inputs. %)
function setupFloatingPlaceholders() {
  const inputs = document.querySelectorAll('input[type="text"][placeholder], input[type="number"][placeholder], input[type="search"][placeholder], textarea[placeholder]');

  inputs.forEach(input => {
    %( Skip placeholders made of non-breaking spaces only. %)
    if (input.placeholder.trim() === '') return;

    let wrapper = input.parentNode;
    if (wrapper.classList.contains('clear-button-wrapper')) {
      %( Reuse the positioned wrapper added by clearinput; keep the
         .input-wrapper CSS hooks so layout rules still match. %)
      wrapper.classList.add('input-wrapper');
    } else if (!wrapper.classList.contains('input-wrapper')) {
      const hadFocus = document.activeElement === input;
      wrapper = document.createElement('div');
      wrapper.className = 'input-wrapper';
      input.parentNode.insertBefore(wrapper, input);
      wrapper.appendChild(input);
      if (hadFocus) requestAnimationFrame(() => input.focus());
    }

    const placeholder = document.createElement('span');
    placeholder.className = 'floating-placeholder';
    placeholder.textContent = input.placeholder;
    wrapper.appendChild(placeholder);

    input.addEventListener('focus', () => placeholder.classList.add('active'));
    input.addEventListener('blur', () => placeholder.classList.remove('active'));

    if (input === document.activeElement || input.hasAttribute('autofocus')) {
      placeholder.classList.add('active');
    }
  });
}
document.addEventListener('DOMContentLoaded', setupFloatingPlaceholders);

function reducePlaces() {
  const seen = [];
  document.querySelectorAll('[data-updfield$="_place"]').forEach(el => {
    const a = el.querySelector('abbr');
    const full = (a ? a.title : el.textContent).trim();
    const raw = (el.dataset.updraw || '').trim();
    const sub = raw.startsWith('%[') && /\%]\s(?:–|—|-)\s/.test(raw);
    const sep = sub ? full.search(/\s(?:–|—|-)\s/) : -1;
    const i = full.indexOf(',', sep < 0 ? 0 : sep);
    if (i > 0 &&
        seen.some(s => full.endsWith(s) && full.length - s.length <= i)) {
      const abbr = document.createElement('abbr');
      abbr.title = full;
      abbr.textContent = full.slice(0, i);
      el.replaceChildren(abbr);
    } else if (a) {
      el.textContent = full;
    }
    seen.push(full);
  });
}
reducePlaces();
