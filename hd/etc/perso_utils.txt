%( ======================================================================
   Affichage d'une personne

   Briques : person_key, person_mod_label, person_sex_icon,
   person_sosa_img, person_sosa_icon, person_label, person_link,
   person_dates_mod.

   Générique : person(xx, p_opt), où p_opt est une suite de lettres :
     i  icône du sexe (lien de modification en mode magicien)
     s  icône Sosa
     c  personne centrale en gras et sans lien
     u  lien souligné à la couleur du sexe
     f  prénom seul si même nom que le père (sauf b.always_surname=yes)
     t  titre principal, avant les dates
     d  dates en italique, lien de modification (magicien), âge en infobulle
     D  dates sans lien
     T  titre principal après les dates, précédé d'une virgule
     b  « joyeux anniversaire »
   Raccourcis : person_name, person_name_dates, person_name_dates_icon,
   person_name_dates_plain, person_firstname_dates.

   Le lien vers une personne porte en infobulle « Prénom.occ Nom » quand
   son nom est visible (xx.is_name_visible, même règle que %xx;).

   Les paramètres sont remplacés par substitution de texte brute : garder
   « xx » pour la personne et des noms préfixés par « p_ » pour le reste.
   ====================================================================== %)

%( Prénom.occ Nom %)
%define;person_key(xx)%xx.first_name;.%xx.occ; %xx.surname;%end;

%define;person_mod_label(xx)[*modify::] %apply;person_key("xx")%end;

%define;person_sex_icon(xx)
  %if;cancel_links;
    %if;xx.is_male;♂%elseif;xx.is_female;♀%else;⚲%end;%sp;
  %else;
    %if;wizard;<a href="%prefix;m=MOD_IND&%xx.access;" title="%apply;person_mod_label("xx")">%end;%nn;
    <i class="fa %if;xx.is_male;fa-mars male%elseif;xx.is_female;fa-venus female%else;fa-neuter neuter%end; fa-xs me-1"></i>%nn;
    %if;wizard;</a>%end;
  %end;
%end;

%define;person_sosa_img(p_ref, p_num)
  <i class="far fa-circle-dot text-success me-1" title="%if;(p_num!=1)[*direct ancestor of %s:::p_ref] – [*Sosa] p_num%end;"></i>%nn;
%end;

%define;person_sosa_icon(xx)
  %if;(xx.has_sosa and b.display_sosa="yes" and self!=sosa_ref)
    %if;not cancel_links;<a href="%prefix;%xx.sosa_link;">%end;
    %apply;person_sosa_img%with;%sosa_ref.first_name_key_strip;%sp;
    %sosa_ref.surname_key_strip;
    %and;%xx.sosa.v;%end;
    %if;not cancel_links;</a>%end;%nn;
  %end;
%end;

%( Nom affiché : %xx;, ou le prénom seul (option f) %)
%define;person_label(xx, p_opt)%nn;
  %if;("f" in "p_opt" and b.always_surname!="yes" and xx.surname=father.surname)%nn;
    %if;(xx.public_name!="")%xx.public_name;%else;%xx.first_name;%end;%nn;
  %else;%xx;%nn;
  %end;
%end;

%( Nom de la personne, avec un lien vers sa fiche sauf si les liens sont
   désactivés ou si la personne est restreinte. Options c, u et f. %)
%define;person_link(xx, p_opt)%nn;
  %if;("c" in "p_opt" and xx.index=central_index)%nn;
    <b>%apply;person_label("xx", "p_opt")</b>%nn;
  %elseif;(cancel_links or xx.is_restricted)%apply;person_label("xx", "p_opt")%nn;
  %else;<a href="%prefix;%xx.access;"%nn;
    %if;xx.is_name_visible; title="%apply;person_key("xx")"%end;%nn;
    %if;("u" in "p_opt") class="%if;(xx.sex=0)male%elseif;(xx.sex=1)female%else;neuter%end;-underline"%end;>%nn;
    %apply;person_label("xx", "p_opt")</a>%nn;
  %end;
%end;

%( Dates en italique ; en mode magicien, lien vers la modification de la
   personne. L'âge est en infobulle. %)
%define;person_dates_mod(xx)
  %if;(cancel_links or not wizard)<span%else;<a href="%prefix;m=MOD_IND&%xx.access;"%end;%sp;
    title="%if;(wizard and not cancel_links)%apply;person_mod_label("xx")%if;(xx.computable_age or xx.computable_death_age) %end;%end;%nn;
      %if;xx.computable_age;(%xx.age;)%elseif;xx.computable_death_age;(%xx.death_age;)%end;" class="text-nowrap fst-italic">%if;(xx.dates!="")%xx.dates;%end;
  %if;(cancel_links or not wizard)</span>%else;</a>%end;
%end;

%define;person(xx, p_opt)
  %if;("i" in "p_opt")%apply;person_sex_icon("xx")%end;
  %if;("s" in "p_opt")%apply;person_sosa_icon("xx")%end;
  %apply;person_link("xx", "p_opt")
  %if;("t" in "p_opt")%xx.title;%end;
  %if;("d" in "p_opt")%apply;person_dates_mod("xx")%end;%nn;
  %if;("D" in "p_opt")<span class="text-nowrap">%xx.dates;</span>%end;%nn;
  %if;("T" in "p_opt" and xx.title!=""), %xx.title;%end;%nn;
  %if;("b" in "p_opt" and xx.birthday) ([happy birthday to you!])%end;%nn;
%end;

%( Nom, titre et dates, sans icône ni mise en gras %)

%( Icône du sexe, Sosa, prénom seul si même nom que le père, dates %)

%( Nom suivi des événements principaux, sur une ligne. Option h : rien
   pour la personne centrale, âge du père ou de la mère à sa naissance. %)
%define;person_summary(xx, p_opt)
  %if;not ("h" in "p_opt" and xx.index=central_index)
    %apply;person_sex_icon("xx")
    %apply;person_sosa_icon("xx")
    %apply;person_link("xx", "c")
    %if;(xx.index!=central_index and xx.has_nobility_titles)
      %foreach;xx.nobility_title;, %nobility_title;%end;
    %end;
  %end;
  <span class="small">%nn;
    %if;("h" in "p_opt" and has_parents)
      %if;(xx.index=father.index or xx.index=mother.index)
        %if;(has_birth_date and xx.has_birth_date)%sp;
          (<abbr title="%apply;age_at_birth_of(central_person)">%xx_age_at_birth;</abbr>)%end;
      %end;
    %end;
    %if;xx.has_birth_date;, %apply;nth([born],xx.sex) %xx.on_birth_date;
      %if;xx.birthday; ([happy birthday to you!])%end;
      %if;xx.has_birth_place; - %xx.birth_place;%end;
    %else;
      %if;xx.has_birth_place;, %apply;nth([born],xx.sex) - %xx.birth_place;%end;
    %end;
    %if;xx.has_baptism_date;, %apply;nth([baptized],xx.sex) %xx.on_baptism_date;
      %if;xx.has_baptism_place; - %xx.baptism_place;%end;
    %else;
      %if;xx.has_baptism_place;, %apply;nth([baptized],xx.sex) - %xx.baptism_place;%end;
    %end;
    %if;xx.computable_age;, [age][:] %xx.age;%end;
    %if;xx.is_dead;, %xx.died;
      %if;xx.has_death_date; %xx.on_death_date;%end;
      %if;xx.has_death_place; - %xx.death_place;%end;
      %if;xx.computable_death_age; [aged] %xx.death_age;%end;
    %end;
    %if;xx.is_buried;, %apply;nth([buried],xx.sex)%sp;
      %if;xx.has_burial_date; %xx.on_burial_date;%end;
      %if;xx.has_burial_place; - %xx.burial_place;%end;
    %end;
    %if;xx.is_cremated;, %apply;nth([cremated],xx.sex)%sp;
      %if;xx.has_cremation_date; %xx.on_cremation_date;%end;
      %if;xx.has_cremation_place; - %xx.cremation_place;%end;
    %end;
    %if;xx.has_occupation;, %xx.occupation;%end;
    %if;(xx.has_consanguinity and b.show_consang!="no"),%sp;
      [consanguinity][:] %xx.consanguinity;%end;
  </span>%nn;
%end;

%( Nom suivi des événements, en liste %)
%define;person_summary_list(xx)
  %if;(xx.has_birth_date or xx.has_birth_place or
       xx.has_baptism_date or xx.has_baptism_place or
       xx.is_dead or xx.is_buried or xx.is_cremated or
       xx.has_event or
      (xx.has_consanguinity and b.show_consang!="no") or
       xx.has_occupation or (xx.has_linked_page.OCCU and xx.linked_page.OCCU !=""))
    <ul>
      %if;(xx.index!=central_index)
        <li>
          %apply;person_sex_icon("xx")
          %apply;person_sosa_icon("xx")
          %apply;person_link("xx", "")%sp;
        </li>%nl;
        %if;xx.has_nobility_titles;
          %foreach;xx.nobility_title;
            <li>%apply;capitalize(nobility_title)</li>%nl;
          %end;
        %end;
      %end;
      %if;xx.has_birth_date;
        <li>%apply;capitalize(nth([born],xx.sex)) %xx.on_birth_date;
        %if;xx.birthday; ([happy birthday to you!])%end;
        %if;xx.has_birth_place; - %xx.birth_place;%end;</li>%nl;
      %else;
        %if;xx.has_birth_place;
          <li>%apply;capitalize(nth([born],xx.sex)) - %xx.birth_place;</li>%nl;
        %end;
      %end;
      %if;xx.has_baptism_date;
        <li>%apply;capitalize(nth([baptized],xx.sex)) %xx.on_baptism_date;
        %if;xx.has_baptism_place; - %xx.baptism_place;%end;</li>%nl;
      %else;
        %if;xx.has_baptism_place;
          <li>%apply;capitalize(nth([baptized],xx.sex)) - %xx.baptism_place;</li>%nl;
        %end;
      %end;
      %if;xx.computable_age;
        <li>[*age] : %xx.age;</li>%nl;
      %end;
      %if;xx.has_event;
        %foreach;xx.event;
          %if;(event.name!=[birth] and
               event.name!=[baptism] and
               event.name!=[marriage event] and
               event.name!=[death] and
               event.name!=[burial])
              <li>%apply;capitalize(event.name) :%nn;
              %if;event.has_date; %event.date;%end;
              %if;event.has_place; - %event.place;%end;</li>%nl;
          %end;
        %end;
      %end;
      %if;xx.is_dead;
        <li>%apply;capitalize(xx.died)
        %if;xx.has_death_date; %xx.on_death_date;%end;
        %if;xx.has_death_place; - %xx.death_place;%end;
        %if;(xx.has_linked_page.DEATH and xx.linked_page.DEATH !="") %xx.linked_page.DEATH;%end;
        %if;xx.computable_death_age; [aged] %xx.death_age;%end;</li>%nl;
        %if;xx.is_buried;
          <li>%apply;capitalize(nth([buried],xx.sex))
          %if;xx.has_burial_date; %xx.on_burial_date;%end;
          %if;xx.has_burial_place; - %xx.burial_place;%end;</li>%nl;
        %end;
        %if;xx.is_cremated;
          <li>%apply;capitalize(nth([cremated],xx.sex))
          %if;xx.has_cremation_date; %xx.on_cremation_date;%end;
          %if;xx.has_cremation_place; - %xx.cremation_place;%end;</li>%nl;
        %end;
      %end;
      %if;(xx.has_occupation or (xx.has_linked_page.OCCU and xx.linked_page.OCCU!=""))
        <li>%if;xx.has_occupation;%apply;capitalize(xx.occupation)%end;%nn;
          %if;(xx.has_linked_page.OCCU and xx.linked_page.OCCU!="")%nn;
            %if;xx.has_occupation;,%sp;%xx.linked_page.OCCU;%else;%apply;capitalize(xx.linked_page.OCCU)%end;
          %end;</li>%nl;
      %end;
      %if;(xx.has_consanguinity and b.show_consang!="no")
        <li>[*consanguinity] : %xx.consanguinity;</li>%nl;
      %end;
    </ul>
  %else;&nbsp;%end;
%end;

    %else;

    %end;
  %else;

  %end;

%( Descendance : cercle sans union, disque avec des unions sans enfant,
   carré avec au moins un enfant %)
%define;li_desc(xx)
  %reset_count3;
  %if;xx.has_families;
    %foreach;xx.family;
      %if;(nb_children>0)%incr_count3;%end;
    %end;
  %end;
  <li style="vertical-align: middle;list-style-type: %nn;
    %if;xx.has_families;%if;(count3!=0)square%else;disc%end;%else;circle%end;;">%nn;
%end;

%define;rel_date_place(sexx, date_place)
  %if;(sexx=1)
    %if;are_married;[married%t to:::date_place]1%nn;
    %elseif;are_not_married;[relationship%t to:::date_place]1%nn;
    %elseif;are_engaged;[engaged%t to:::date_place]1%nn;
    %elseif;is_no_sexes_check;[relationship%t to:::date_place]1%nn;
    %elseif;are_pacs;[pacs%t to:::date_place]1%nn;
    %elseif;is_no_mention;[no mention] | date_place [with]%nn;
    %elseif;are_marriage_banns;[marriage banns%t to:::date_place]1%nn;
    %elseif;are_marriage_contract;[marriage contract%t with:::date_place]1%nn;
    %elseif;are_marriage_license;[married%t to:::date_place]1%nn;
    %elseif;are_pacs;[pacsed%t to:::date_place]1%nn;
    %elseif;are_residence;[residence%t to:::date_place]1%nn;
    %end;
  %else;
    %if;are_married;[married%t to:::date_place]0%nn;
    %elseif;are_not_married;[relationship%t to:::date_place]0%nn;
    %elseif;are_engaged;[engaged%t to:::date_place]0%nn;
    %elseif;is_no_sexes_check;[relationship%t to:::date_place]0%nn;
