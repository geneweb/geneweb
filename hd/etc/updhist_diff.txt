function updateRadios(changed) {
  const newEl = document.querySelector('input[name=new]:checked');
  const oldEl = document.querySelector('input[name=old]:checked');
  if (!newEl || !oldEl) return;
  const newVal = parseInt(newEl.value, 10);
  const oldVal = parseInt(oldEl.value, 10);
  if (changed === 'new' && newVal >= oldVal) {
    const target = document.querySelector(`input[name=old][value="${newVal + 1}"]`);
    if (target) target.checked = true;
  } else if (changed === 'old' && oldVal <= newVal) {
    const target = document.querySelector(`input[name=new][value="${oldVal - 1}"]`);
    if (target) target.checked = true;
  }
}
</script>
